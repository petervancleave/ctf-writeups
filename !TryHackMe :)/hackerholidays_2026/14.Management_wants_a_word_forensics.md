Web Data - contains autofill information

`find C/Users/vera -type f \( -iname 'Login Data' -o -iname 'Local State' -o -iname 'Web Data' -o -iname 'History' \) -print`

Now decrypt Chromes AES Key:

`
python3 -c '
from impacket.dpapi import DPAPI_BLOB
masterkey = bytes.fromhex("5e5715ec9b6df5a86e97902692a66d28e691f05d5bc1e04d0159cfe960e94c978c07e5004a0179d3a96df2468885a28175b0b02cc064445f116a752d2b3e9d40")
with open("chrome-key.dpapi", "rb") as f:
    blob = DPAPI_BLOB(f.read())
key = blob.decrypt(masterkey)
if key is None:
    print("Decryption failed")
else:
    print("Chrome AES key:", key.hex())
    open("chrome-aes.key", "wb").write(key)
    print("Saved → chrome-aes.key")
'

`

Now decrypt the password:

`

python3 -c '
import sqlite3, sys
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
key = open("chrome-aes.key","rb").read()
db = sys.argv[1]
conn = sqlite3.connect(f"file:{db}?mode=ro", uri=True)
for url, user, enc in conn.execute("SELECT origin_url, username_value, password_value FROM logins"):
    b = bytes(enc)
    if not b.startswith((b"v10", b"v11")): continue
    nonce, ct = b[3:15], b[15:]
    try:
        pw = AESGCM(key).decrypt(nonce, ct, None).decode()
        print("URL     :", url)
        print("Username:", user)
        print("Password:", pw)
    except Exception as e:
        print("Failed:", e)
' "/home/kali/Desktop/management-wants-a-word-forensics-hh-day-14/KAPE/C/Users/vera/AppData/Local/Google/Chrome For Testing/User Data/Default/Login Data"
